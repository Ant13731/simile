
% Comprehensions
%% Set
\makeatletter
\newcommand{\Set}[1]{%
  \@ifnextchar\bgroup
    {\Set@check{#1}}
    {\{\,#1\,\}}%
}
\def\Set@check#1#2{%
  \@ifnextchar\bgroup
    {\Set@three{#1}{#2}}
    {\{\,#1 \mid #2\,\}}%
}
\def\Set@three#1#2#3{%
  \{\,#1 \cdot #2 \mid #3\,\}%
}
\makeatother

%% Bag
\makeatletter
\newcommand{\Bag}[1]{%
  \@ifnextchar\bgroup
    {\Bag@check{#1}}
    {\llbracket\,#1\,\rrbracket}%
}
\def\Bag@check#1#2{%
  \@ifnextchar\bgroup
    {\Bag@three{#1}{#2}}
    {\llbracket\,#1 \mid #2\,\rrbracket}%
}
\def\Bag@three#1#2#3{%
  \llbracket\,#1 \cdot #2 \mid #3\,\rrbracket%
}
\makeatother

% For footnotes within equations
% From https://tex.stackexchange.com/questions/82782/footnote-in-align-environment
\makeatletter
\let\original@footnote\footnote
\newcommand{\align@footnote}[1]{%
  \ifmeasuring@
    \chardef\@tempfn=\value{footnote}%
    \footnotemark
    \setcounter{footnote}{\@tempfn}%
  \else
    \iffirstchoice@
      \original@footnote{#1}%
    \fi
  \fi}
\pretocmd{\start@align}{\let\footnote\align@footnote}{}{}
\makeatother
