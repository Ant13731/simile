
% Comprehensions
\makeatletter
% \Comprehension{left}{right}{expr}
\newcommand{\Comprehension}[3]{%
  \@ifnextchar\bgroup
    {\Comprehension@check{#1}{#2}{#3}}
    {#1\,#3\,#2}%
}
\def\Comprehension@check#1#2#3#4{%
  \@ifnextchar\bgroup
    {\Comprehension@three{#1}{#2}{#3}{#4}}
    {#1\,#3 \mid #4\,#2}%
}
\def\Comprehension@three#1#2#3#4#5{%
  #1\,#3 \cdot #4 \mid #5\,#2%
}
\makeatother

% Quantifier
\makeatletter
\newcommand{\Quantifier}[2]{%
  \@ifnextchar\bgroup
    {\Quantifier@three{#1}{#2}}
    {\,#1\,#2}%
}
\newcommand{\Quantifier@three}[3]{%
  \@ifnextchar\bgroup
    {\Quantifier@four{#1}{#2}{#3}}
    {\,#1\,#2 \mid #3}%
}
\newcommand{\Quantifier@four}[4]{%
  \,#1\,#2 \cdot #3 \mid #4%
}
\makeatother

% For footnotes within equations
% From https://tex.stackexchange.com/questions/82782/footnote-in-align-environment
\makeatletter
\let\original@footnote\footnote
\newcommand{\align@footnote}[1]{%
  \ifmeasuring@
    \chardef\@tempfn=\value{footnote}%
    \footnotemark
    \setcounter{footnote}{\@tempfn}%
  \else
    \iffirstchoice@
      \original@footnote{#1}%
    \fi
  \fi}
\pretocmd{\start@align}{\let\footnote\align@footnote}{}{}
\makeatother
